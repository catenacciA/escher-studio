cmake_minimum_required(VERSION 3.20)
project(EscherStudio VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(escher_core src/Core.cpp src/Art.cpp)
target_include_directories(escher_core PUBLIC src)
target_compile_options(escher_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/CoreTests.cpp)
  target_link_libraries(core_tests PRIVATE escher_core)
  add_test(NAME mathematics COMMAND core_tests)
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0")
  add_executable(EscherStudio MACOSX_BUNDLE src/App.mm src/Renderer.cpp resources/Deform.metal)
  target_sources(EscherStudio PRIVATE LICENSE metal-cpp/LICENSE.txt)
  set_source_files_properties(LICENSE PROPERTIES MACOSX_PACKAGE_LOCATION Resources/Licenses)
  set_source_files_properties(metal-cpp/LICENSE.txt PROPERTIES MACOSX_PACKAGE_LOCATION Resources/Licenses/Metal-cpp)
  set_source_files_properties(resources/Deform.metal PROPERTIES MACOSX_PACKAGE_LOCATION Resources HEADER_FILE_ONLY TRUE)
  # Third-party artwork textures and their manifest (see resources/artwork/README.md).
  file(GLOB ARTWORK resources/artwork/*.jpg resources/artwork/*.json resources/artwork/*.md)
  target_sources(EscherStudio PRIVATE ${ARTWORK})
  set_source_files_properties(${ARTWORK} PROPERTIES MACOSX_PACKAGE_LOCATION Resources/artwork HEADER_FILE_ONLY TRUE)
  target_include_directories(EscherStudio PRIVATE metal-cpp)
  target_compile_options(EscherStudio PRIVATE -Wall -Wextra -fblocks "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(EscherStudio PRIVATE escher_core "-framework AppKit" "-framework Metal" "-framework MetalKit" "-framework QuartzCore" "-framework ImageIO" "-framework CoreGraphics" "-framework UniformTypeIdentifiers")
  set_target_properties(EscherStudio PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist" MACOSX_BUNDLE_BUNDLE_NAME "Escher Studio" MACOSX_BUNDLE_GUI_IDENTIFIER "studio.escher.tile" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3.0")
endif()
